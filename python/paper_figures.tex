% ---------- Section 4.2 : the admissibility mechanism (inner-term variance) ----------
\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{mechanism_single_state}
  \caption{\textbf{Inner-term estimator variance (single state).}
  The admissible $\Omega$ (RKL) has constant integrand $\Phi_{\mathrm{KL}}\!\equiv\!1$, so its
  Monte-Carlo estimate $\hat{C}_\Omega=\tfrac1n\sum_i\Phi(u_{a_i})$ has \emph{exactly} zero
  variance at every budget $n$; every non-admissible $\Omega$ decays only as $1/\sqrt{n}$.}
  \label{fig:inner-single-state}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{mechanism_trajectory}
  \caption{\textbf{Inner-term noise accumulates over a trajectory.}
  A horizon-$H$ rollout sums $H\!-\!1$ inner-term draws: the per-trajectory std grows as
  $\sqrt{H-1}$ for non-admissible $\Omega$, while the admissible $\Omega$ stays exactly $0$.}
  \label{fig:inner-trajectory}
\end{figure}

% ---------- Section 4.3 : policy recovery ----------
\begin{figure*}[t]
  \centering
  \includegraphics[width=\linewidth]{tabular_headline_p70}
  \caption{\textbf{Policy recovery across data regimes.}
  Final gap $\Delta\pi=\mathrm{TV}(\pi_\theta,\pi^\star_\Omega)$ vs.\ Monte-Carlo budget
  $n_{\mathrm{mc}}$ (calibration peak $0.7$). Only the admissible $\Omega$ (RKL) recovers
  $\pi^\star$ at low/zero budget; the others require large $n_{\mathrm{mc}}$.
  $100$ independent MDPs, $\pm95\%$ CI.}
  \label{fig:headline}
\end{figure*}

\begin{figure}[t]
  \centering
  \includegraphics[width=0.92\linewidth]{tabular_offpolicy_peaks}
  \caption{\textbf{Off-policy recovery vs.\ calibration peak.}
  RKL attains the lowest off-policy gap $\Delta\pi$ at every target peak, and its margin over the
  non-admissible $\Omega$ widens as $\pi^\star$ sharpens ($0.6\!\to\!0.8$).}
  \label{fig:peak-ablation}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=0.85\linewidth]{tabular_eps_offpolicy}
  \caption{\textbf{Off-policy recovery vs.\ transition noise $\varepsilon$.}
  With $\alpha$ held fixed at the $\varepsilon{=}0.2$ calibration (so only the dynamics vary),
  RKL stays lowest at every $\varepsilon\in[0,0.5]$: admissibility is robust to transition
  stochasticity.}
  \label{fig:eps-ablation}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=0.8\linewidth]{tabular_alpha_sweep}
  \caption{\textbf{Calibration.} Peakiness $\mathrm{mean}_s\max_a\pi^\star(a\mid s)$ as a function
  of the regularization weight $\alpha$ (MDP $0$, deterministic). Each $\Omega$'s $\alpha$ is set
  to a common target peak (markers), so comparisons are not confounded by $\alpha$.}
  \label{fig:alpha-calib}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{tabular_ci_n10_vs_n100}
  \caption{\textbf{Effect of the number of MDPs.}
  Off-policy $\Delta\pi$ at peak $0.7$ with $n{=}10$ vs.\ $n{=}100$ MDPs. The CI shrinks by
  $\approx\!\sqrt{10}$, but the RKL-vs-rest separation already holds at $n{=}10$.}
  \label{fig:ci-nmdp}
\end{figure}
